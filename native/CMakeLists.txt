cmake_minimum_required(VERSION 3.25)
project(YubiLatchCredentialProvider LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "YubiLatch requires Windows x64")
endif()
add_library(YubiLatch.CredentialProvider SHARED Provider.cpp Provider.def)
target_compile_features(YubiLatch.CredentialProvider PRIVATE cxx_std_20)
target_compile_definitions(YubiLatch.CredentialProvider PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(YubiLatch.CredentialProvider PRIVATE /W4 /WX /sdl /guard:cf /utf-8 /EHsc)
target_link_options(YubiLatch.CredentialProvider PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /CETCOMPAT)
set_property(TARGET YubiLatch.CredentialProvider PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(YubiLatch.CredentialProvider PRIVATE credui secur32 advapi32 shlwapi ole32 user32 gdi32)
